#} {# ====================================================================== Partial 7: Using django-template-partials inline partials ====================================================================== django-template-partials lets you define named sections within a template using {% partialdef name %}...{% endpartialdef %}. The view can then render just that section by appending #name to the template path: render(request, "template.html#section-name", context). The "inline" keyword on partialdef means the partial is also rendered in place when the full template loads. Without "inline", the partial is only rendered when explicitly requested. This avoids maintaining separate partial files for content that is tightly coupled to the full page template. ====================================================================== #} {# --- content/essay_list.html (full template with inline partials) --- #} {% extends "base.html" %} {% load partials %} {% block content %}
